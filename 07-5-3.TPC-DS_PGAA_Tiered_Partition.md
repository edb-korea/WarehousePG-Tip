## TPC-DS PGAA, Iceberf, Tiered Partition Table 구성 

아래는 TPC-DS의 Dimension data 는 DB Table에, Fact Data는 DB Table로 약 25%, PGAA/PGFS로 S3에 Iceberg Table로 약 75% 구성하는 방법을 기술하고 있습니다.

## 0. 사전 구성 사항
1. TPC-DS 환경이 구성되고 DB에 schema 'tpcds'가 'tpcds_org' 로 변경
```sql
ALTER SCHEMA tpcds RENAME TO tpcds_org;
```

2. PGAA/PGFS S3(Minio) 환경이 구성되어 있어야 함.

## 1. 기존 TPC-DS 스키마에서 Tiered Partition Table 로 구성하는 DDL/DML SQL 생성.
```bash
DBNAME=tpcds_sf_10 \
SOURCE_SCHEMA=tpcds_org \
TARGET_SCHEMA=tpcds \
STORAGE_LOCATION=minio_pgaa_tpcds \
OUT_FILE=pgaa_s3_tiered_partition_tpcds_all.sql \
bash generate_pgaa_s3_tiered_partition_ddl.sh
```

## 2. DDL/DML SQL 로 신규 schema 구성.
```bash
psql -d tpcds_sf_10 -v ON_ERROR_STOP=1 -f pgaa_s3_tiered_partition_tpcds_all.sql
```

## 3. schema에 대한 권한 부여.
```sql
GRANT USAGE ON SCHEMA tpcds TO dsbench;
GRANT SELECT ON ALL TABLES IN SCHEMA tpcds TO dsbench;
ALTER DEFAULT PRIVILEGES IN SCHEMA tpcds GRANT SELECT ON TABLES TO dsbench;
```

## 4. TPC-DS 수행.
```bash
nohup ./tpcds.sh &
```

## 5. (옵션) TPC-DS 수행 후. 재 사용을 위해 스키마면 변경.
```sql
alter schema tpdcs rename to tpcds_pgaa_tierd;
```

## 첨부. generate_pgaa_s3_tiered_partition_ddl.sh
```bash
#!/usr/bin/env bash
#
# generate_pgaa_s3_tiered_partition_ddl.sh
#
# Builds a Tiered Partition Table version of the TPC-DS schema in a new
# target schema, using PGAA/PGFS (Iceberg on S3/MinIO) for the "cold" tier:
#
#   - Dimension tables (source relkind = 'r', i.e. not partitioned) are
#     copied as-is into plain native DB tables in the target schema.
#   - Fact tables (source relkind = 'p', i.e. partitioned) are rebuilt as a
#     SINGLE parent partition table per table, whose leaf partitions are a
#     MIX of:
#       * native DB partitions (the "hot" tier) -- NATIVE_PCT% of the
#         source table's leaf partitions, created with
#         CREATE TABLE ... PARTITION OF ... and loaded with a plain INSERT.
#       * PGAA/Iceberg partitions (the "cold" tier) -- the remaining
#         partitions, each written to S3/MinIO with the same two-step CTAS
#         + read-mapped-table pattern as 07-5 (a CTAS write-once table
#         streams the data to object storage, then a read-mapped table with
#         an explicit DISTRIBUTED BY becomes the actual leaf partition),
#         then attached back with ATTACH PARTITION -- matching 04-5's
#         Tiered Partition Table PGAA pattern, just automated across every
#         fact table's partitions instead of built by hand.
#     Queries against the parent table transparently span both tiers.
#
# This is the PGAA/S3 counterpart of generate_pxf_file_tiered_partition_ddl.sh
# and generate_pxf_s3_tiered_partition_ddl.sh (which use PXF external tables
# for the cold tier instead) -- the tiering split logic (NATIVE_PCT /
# NATIVE_TIER) is identical; only how the cold tier is built differs.
#
# Required environment variables:
#   DBNAME            Target database name.
#   SOURCE_SCHEMA     Source TPC-DS schema (e.g. tpcds).
#   TARGET_SCHEMA     New schema to create the tiered structure in.
#   STORAGE_LOCATION  PGFS storage location name registered with
#                     pgfs.create_storage_location() (see 07-5 for setup).
#
# Optional environment variables:
#   OUT_FILE        Output SQL file (default: pgaa_s3_tiered_partition_tpcds_all.sql).
#   NATIVE_PCT      Percent of each fact table's leaf partitions to keep as
#                   native DB partitions (default: 25). The rest go to the
#                   PGAA/Iceberg cold tier. Split is by partition COUNT, not
#                   by row count, so the actual row-level ratio is only
#                   approximate (TPC-DS date partitions are roughly
#                   equal-sized, so it tracks closely in practice). Integer
#                   division truncates, so small partition counts round down.
#   NATIVE_TIER     "newest" (default) or "oldest" -- which partitions stay
#                   native. "newest" keeps the most recent partitions (by
#                   the partition's RANGE lower bound) on fast native
#                   storage and pushes older/colder partitions to
#                   PGAA/Iceberg -- the standard hot/cold tiered-storage
#                   pattern. Set to "oldest" to invert this if needed.
#
# Naming convention in TARGET_SCHEMA:
#   <table_name>            Dimension table (plain copy), or fact table
#                           parent partition table.
#   <child_partition_name>  Leaf partition, same name as the source child
#                           partition -- used for BOTH tiers (native DB
#                           partition or PGAA read-mapped partition); which
#                           one it is only shows up in the comment above it
#                           and in pgaa.table_mapping.
#   w_<child_partition_name>    Transient PGAA CTAS write table used only to
#                           stream a cold partition's data to S3/MinIO;
#                           dropped again right after the matching
#                           read-mapped <child_partition_name> table is
#                           created, so it does not persist in the schema.
#
set -euo pipefail

OUT_FILE="${OUT_FILE:-pgaa_s3_tiered_partition_tpcds_all.sql}"
COUNT_FILE="$(mktemp)"
trap 'rm -f "${COUNT_FILE}"' EXIT

: "${DBNAME:?Set DBNAME to the target database name.}"
: "${SOURCE_SCHEMA:?Set SOURCE_SCHEMA to the source TPC-DS schema name.}"
: "${TARGET_SCHEMA:?Set TARGET_SCHEMA to the target schema name.}"
: "${STORAGE_LOCATION:?Set STORAGE_LOCATION to the PGFS storage location name.}"

NATIVE_PCT="${NATIVE_PCT:-25}"
NATIVE_TIER="${NATIVE_TIER:-newest}"

if [[ "${NATIVE_TIER}" != "newest" && "${NATIVE_TIER}" != "oldest" ]]; then
  echo "ERROR: NATIVE_TIER must be 'newest' or 'oldest' (got '${NATIVE_TIER}')" >&2
  exit 1
fi

if ! [[ "${NATIVE_PCT}" =~ ^[0-9]+$ ]] || (( NATIVE_PCT < 0 || NATIVE_PCT > 100 )); then
  echo "ERROR: NATIVE_PCT must be an integer between 0 and 100 (got '${NATIVE_PCT}')" >&2
  exit 1
fi

psql -d "${DBNAME}" -v ON_ERROR_STOP=1 -Atc "select 1" >/dev/null

cat > "${OUT_FILE}" <<SQL
\\set ON_ERROR_STOP on

CREATE EXTENSION IF NOT EXISTS pgaa CASCADE;
CREATE SCHEMA IF NOT EXISTS ${TARGET_SCHEMA};

SQL

cat > "${COUNT_FILE}" <<SQL

--
-- Row count validation
--
SQL

# Drops an object and cleans up its pgaa.table_mapping row, if any. Safe to
# call on any object name in TARGET_SCHEMA regardless of whether it is (or
# ever was) a PGAA table -- the DELETE simply affects 0 rows otherwise. This
# is called before (re-)creating every object so reruns are safe even if a
# child partition's tier (native vs PGAA) changed between runs.
emit_drop_pgaa_table() {
  local object_name="$1"

  cat >> "${OUT_FILE}" <<SQL
DROP TABLE IF EXISTS ${TARGET_SCHEMA}.${object_name} CASCADE;
DELETE FROM pgaa.table_mapping
 WHERE rel_namespace = '${TARGET_SCHEMA}'
   AND rel_name = '${object_name}';
SQL
}

psql -d "${DBNAME}" -v ON_ERROR_STOP=1 -At -F '|' <<SQL | while IFS='|' read -r table_name relkind dist_clause part_key column_defs
SELECT
    c.relname AS table_name,
    c.relkind,
    COALESCE(pg_get_table_distributedby(c.oid), 'DISTRIBUTED RANDOMLY') AS dist_clause,
    COALESCE(pg_get_partkeydef(c.oid), '') AS part_key,
    (
        SELECT string_agg(
            format(
                '%I %s',
                a.attname,
                pg_catalog.format_type(a.atttypid, a.atttypmod)
            ),
            ', '
            ORDER BY a.attnum
        )
        FROM pg_attribute a
        WHERE a.attrelid = c.oid
          AND a.attnum > 0
          AND NOT a.attisdropped
    ) AS column_defs
FROM pg_class c
JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE n.nspname = '${SOURCE_SCHEMA}'
  AND c.relkind IN ('r', 'p')
  AND NOT EXISTS (
      SELECT 1
      FROM pg_inherits i
      WHERE i.inhrelid = c.oid
  )
ORDER BY c.relname;
SQL
do
  if [[ -z "${column_defs}" ]]; then
    echo "ERROR: column definitions were not found for ${SOURCE_SCHEMA}.${table_name}" >&2
    exit 1
  fi

  cat >> "${COUNT_FILE}" <<SQL

SELECT '${table_name}' AS table_name,
       (SELECT count(*) FROM ${SOURCE_SCHEMA}.${table_name}) AS source_count,
       (SELECT count(*) FROM ${TARGET_SCHEMA}.${table_name}) AS target_count;
SQL

  if [[ "${relkind}" == "r" ]]; then
    # --- Dimension table: plain copy into the target schema, no tiering. ---
    cat >> "${OUT_FILE}" <<SQL

--
-- Dimension table (plain copy, no tiering): ${SOURCE_SCHEMA}.${table_name}
--
DROP TABLE IF EXISTS ${TARGET_SCHEMA}.${table_name} CASCADE;

CREATE TABLE ${TARGET_SCHEMA}.${table_name}
AS
SELECT *
FROM ${SOURCE_SCHEMA}.${table_name}
${dist_clause};
SQL
    continue
  fi

  # --- relkind = 'p': fact table -> Tiered Partition Table. ---
  if [[ -z "${part_key}" ]]; then
    echo "ERROR: partition key was not found for ${SOURCE_SCHEMA}.${table_name}" >&2
    exit 1
  fi

  cat >> "${OUT_FILE}" <<SQL

--
-- Fact table (Tiered Partition Table): ${SOURCE_SCHEMA}.${table_name}
-- Native (hot) tier: ${NATIVE_PCT}% of leaf partitions, ${NATIVE_TIER}
-- External (cold) tier: remaining leaf partitions, via PGAA/Iceberg on S3/MinIO
--
SQL
  emit_drop_pgaa_table "${table_name}"

  cat >> "${OUT_FILE}" <<SQL

CREATE TABLE ${TARGET_SCHEMA}.${table_name}
(${column_defs})
${dist_clause}
PARTITION BY ${part_key};
SQL

  # Collect every leaf partition of this source table, ordered oldest ->
  # newest by the numeric lower bound of its RANGE partition. Using
  # process substitution (not a pipe) here means the resulting bash array
  # survives in the current shell, so the total count is known up front and
  # the NATIVE_PCT / NATIVE_TIER split can be computed before emitting SQL.
  mapfile -t child_rows < <(
    psql -d "${DBNAME}" -v ON_ERROR_STOP=1 -At -F '|' <<SQL
SELECT
    child.relname,
    pg_get_expr(child.relpartbound, child.oid),
    COALESCE(
        (regexp_match(pg_get_expr(child.relpartbound, child.oid), 'FROM \(([0-9]+)'))[1]::bigint,
        0
    ) AS lower_bound
FROM pg_inherits i
JOIN pg_class parent ON parent.oid = i.inhparent
JOIN pg_namespace parent_ns ON parent_ns.oid = parent.relnamespace
JOIN pg_class child ON child.oid = i.inhrelid
JOIN pg_namespace child_ns ON child_ns.oid = child.relnamespace
WHERE parent_ns.nspname = '${SOURCE_SCHEMA}'
  AND parent.relname = '${table_name}'
  AND child_ns.nspname = '${SOURCE_SCHEMA}'
ORDER BY lower_bound;
SQL
  )

  total_children="${#child_rows[@]}"
  native_count=$(( total_children * NATIVE_PCT / 100 ))

  for idx in "${!child_rows[@]}"; do
    IFS='|' read -r child_name part_bound lower_bound <<< "${child_rows[$idx]}"

    # child_rows is ordered oldest -> newest (ascending lower_bound).
    if [[ "${NATIVE_TIER}" == "newest" ]]; then
      if (( idx >= total_children - native_count )); then
        is_native=1
      else
        is_native=0
      fi
    else
      if (( idx < native_count )); then
        is_native=1
      else
        is_native=0
      fi
    fi

    if [[ "${is_native}" == "1" ]]; then
      emit_drop_pgaa_table "${child_name}"

      cat >> "${OUT_FILE}" <<SQL

-- Native (hot) partition: ${SOURCE_SCHEMA}.${child_name}
-- ${part_bound}
CREATE TABLE ${TARGET_SCHEMA}.${child_name}
PARTITION OF ${TARGET_SCHEMA}.${table_name}
${part_bound};

INSERT INTO ${TARGET_SCHEMA}.${child_name}
SELECT *
FROM ${SOURCE_SCHEMA}.${child_name};
SQL
    else
      pgaa_path="tpcds_tiered/${table_name}/${child_name}"

      cat >> "${OUT_FILE}" <<SQL

-- External (cold) partition: ${SOURCE_SCHEMA}.${child_name}
-- ${part_bound}

-- Step 1: CTAS write -- creates a transient PGAA table and streams its data
-- to object storage in one atomic, write-once operation.
SQL
      emit_drop_pgaa_table "w_${child_name}"

      cat >> "${OUT_FILE}" <<SQL

CREATE TABLE ${TARGET_SCHEMA}.w_${child_name}
USING PGAA
WITH (
    pgaa.storage_location = '${STORAGE_LOCATION}',
    pgaa.path = '${pgaa_path}',
    pgaa.format = 'iceberg',
    pgaa.purge_data_if_exists = true
) AS
SELECT * FROM ${SOURCE_SCHEMA}.${child_name};

SELECT count(*) FROM ${TARGET_SCHEMA}.w_${child_name};
DROP TABLE ${TARGET_SCHEMA}.w_${child_name};

-- Step 2: read-mapped table -- this becomes the actual leaf partition, with
-- an explicit DISTRIBUTED BY matching the parent (CTAS write tables cannot
-- take a user-specified DISTRIBUTED BY, so it has to be a separate table).
SQL
      emit_drop_pgaa_table "${child_name}"

      cat >> "${OUT_FILE}" <<SQL

CREATE TABLE ${TARGET_SCHEMA}.${child_name}
(${column_defs})
USING PGAA
WITH (
    pgaa.storage_location = '${STORAGE_LOCATION}',
    pgaa.path = '${pgaa_path}',
    pgaa.format = 'iceberg'
)
${dist_clause};

ALTER TABLE ${TARGET_SCHEMA}.${table_name}
ATTACH PARTITION ${TARGET_SCHEMA}.${child_name}
${part_bound};
SQL
    fi
  done
done

cat "${COUNT_FILE}" >> "${OUT_FILE}"

echo "Generated: ${OUT_FILE}"
```
