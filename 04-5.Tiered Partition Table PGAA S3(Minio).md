# WarehousePG Tiered Partition 구성 가이드 (S3 오프로드 및 재연결)

WarehousePG 자체에는 "Tiered Partition"이라는 완전 자동화 기능은 없습니다. 자동 티어링(`pgaa.convert_to_tiered_table()`)은 **PGD(EDB Postgres Distributed) + PGAA** 조합에서만 지원되는 기능이고, WarehousePG 단독 환경에서는 **선언적 파티셔닝의 DETACH/ATTACH**와 **PGAA(Postgres Analytics Accelerator)의 CTAS/스토리지 로케이션**을 조합해 수동으로 구성해야 합니다. 아래는 실제 동작하는 예시입니다 (WarehousePG 7.4+, PGAA 필요).

## 0. 사전 준비 — PGAA 확장 및 S3 스토리지 로케이션

```sql
CREATE EXTENSION IF NOT EXISTS pgaa;

-- S3 버킷을 가리키는 storage location 등록 (자격증명 필요)
SELECT pgfs.create_storage_location(
  'sales_cold_tier',
  's3://my-warehouse-archive/sales',
  '{"aws_region": "us-east-1", "aws_access_key_id": "...", "aws_secret_access_key": "..."}'
);

-- Minio 버킷에 대한 연결 정보 생성   
SELECT pgfs.create_storage_location(
    'my_minio_storage',
    's3://pgaa', 
    '{"endpoint": "http://10.0.8.192:9000", 
       "allow_http": "true"}',
    '{"access_key_id": "minioadmin",
      "secret_access_key": "minioadmin"}'        
);

-- S3 연결 정보 확인 및 수정 
SELECT * FROM pgfs.list_storage_locations();

SELECT * FROM pgfs.get_storage_location('my_minio_storage');

## 1. 파티션 테이블 생성 (일반 range partition)

```sql
-- Parent partitioned table
-- No NOT NULL / CHECK constraints → required for EXCHANGE PARTITION with ext table
DROP TABLE public.events_pgaa CASCADE;
CREATE TABLE public.events_pgaa (
    event_id    BIGINT,
    tenant_id   INT,
    event_type  VARCHAR(64),
    payload     TEXT,
    region      TEXT,
    amount      NUMERIC(12, 2),
    created_at  TIMESTAMP
)
WITH (
  appendoptimized = true,
  orientation = column,
  compresstype = zstd,
  compresslevel = 5
)
DISTRIBUTED BY (tenant_id)
PARTITION BY RANGE (created_at)
(
    PARTITION p202506
        START (TIMESTAMP '2025-06-01') INCLUSIVE
        END   (TIMESTAMP '2025-07-01') EXCLUSIVE,
    PARTITION p202505
        START (TIMESTAMP '2025-05-01') INCLUSIVE
        END   (TIMESTAMP '2025-06-01') EXCLUSIVE,
    PARTITION p202504
        START (TIMESTAMP '2025-04-01') INCLUSIVE
        END   (TIMESTAMP '2025-05-01') EXCLUSIVE,
    PARTITION p202503
        START (TIMESTAMP '2025-03-01') INCLUSIVE
        END   (TIMESTAMP '2025-04-01') EXCLUSIVE,
    PARTITION p202502
        START (TIMESTAMP '2025-02-01') INCLUSIVE
        END   (TIMESTAMP '2025-03-01') EXCLUSIVE,
    PARTITION p202501
        START (TIMESTAMP '2025-01-01') INCLUSIVE
        END   (TIMESTAMP '2025-02-01') EXCLUSIVE
);
-- ... 이후 월별로 계속 생성
```

###  Insert sample rows into all partitions

```sql
-- 1000,000 rows spread across Jan–Jun 2025
INSERT INTO events_pgaa (event_id, tenant_id, event_type, payload,
                    region, amount, created_at)
SELECT
    gs                                                             AS event_id,
    (gs % 5) + 1                                                   AS tenant_id,
    (ARRAY['purchase','view','click','signup','refund'])[(gs%5)+1]  AS event_type,
    '{"source":"lab","v":' || gs || '}'                            AS payload,
    (ARRAY['us-east','us-west','eu-west','ap-seoul','ap-tokyo'])[(gs%5)+1]
                                                                   AS region,
    ROUND((RANDOM() * 500 + 10)::NUMERIC, 2)                      AS amount,
    TIMESTAMP '2025-01-01'
        + ((gs - 1) % 181) * INTERVAL '1 day'
        + (gs % 86400)     * INTERVAL '1 second'                  AS created_at
FROM generate_series(1, 1000000) gs;
```

```sql
-- List all partitions with their boundary and storage type
SELECT
    pt.relid::regclass                          AS partition_table,
    pt.level,
    pt.isleaf,
    pg_get_expr(c.relpartbound, c.oid)          AS partition_bound,
    CASE c.relkind
        WHEN 'r' THEN 'heap (local)'
        WHEN 'f' THEN 'external (PXF)'
    END                                         AS storage_type
FROM   pg_partition_tree('events_pgaa') pt
JOIN   pg_class c ON c.oid = pt.relid
WHERE  pt.isleaf = true
ORDER  BY partition_bound;
```

```sql
-- Direct count per partition child table
SELECT 'p202501' AS part, COUNT(*) FROM events_pgaa_1_prt_p202501
UNION ALL
SELECT 'p202502',          COUNT(*) FROM events_pgaa_1_prt_p202502
UNION ALL
SELECT 'p202503',          COUNT(*) FROM events_pgaa_1_prt_p202503
UNION ALL
SELECT 'p202504',          COUNT(*) FROM events_pgaa_1_prt_p202504
UNION ALL
SELECT 'p202505',          COUNT(*) FROM events_pgaa_1_prt_p202505
UNION ALL
SELECT 'p202506',          COUNT(*) FROM events_pgaa_1_prt_p202506;
```

## 2. 오래된 파티션을 S3로 내보내기 (Cold tier로 이동)

먼저 파티션 계층에서 분리(DETACH)한 뒤, PGAA CTAS로 S3에 Iceberg/Delta 포맷으로 써냅니다.

```sql
-- ① 파티션 분리 (부모 테이블에서 떼어내지만 테이블 자체는 그대로 존재)
ALTER TABLE events_pgaa DETACH PARTITION events_pgaa_1_prt_p202501;

-- ② PGAA로 S3에 export (CTAS는 write-once, 원자적 동작)
DROP TABLE IF EXISTS public.events_pgaa_1_prt_p202501_write;
CREATE TABLE public.events_pgaa_1_prt_p202501_write
USING PGAA
WITH (
    pgaa.storage_location = 'my_minio_storage',
    pgaa.path = 'events/2025/01',
    pgaa.format = 'iceberg',
    pgaa.purge_data_if_exists = true
)
-- DISTRIBUTED BY (tenant_id)
AS
SELECT * FROM public.events_pgaa_1_prt_p202501;

-- pgaa write table 확인 후, 테이블 삭제. 삭제해도 데이타는 Object Storage에 그대로 있음.
SELECT count(*) FROM public.events_pgaa_1_prt_p202501_write; -- row count 확인
DROP TABLE IF EXISTS public.events_pgaa_1_prt_p202501_write;

-- pgaa read table 생성., 기존 pgaa write table로 만들어진 데이터로.
DROP TABLE IF EXISTS public.events_pgaa_1_prt_p202501_read;
CREATE TABLE public.events_pgaa_1_prt_p202501_read( LIKE events_pgaa_1_prt_p202501 )
USING PGAA
WITH (
    pgaa.storage_location = 'my_minio_storage',
    pgaa.path = 'events/2025/01',
    pgaa.format = 'iceberg'
)
DISTRIBUTED BY (tenant_id);

SELECT count(*) FROM public.events_pgaa_1_prt_p202501_read; -- row count 확인

-- 단일 레벨 파티션: ALTER PARTITION 없이 바로 EXCHANGE PARTITION
ALTER TABLE events_pgaa
    EXCHANGE PARTITION p202501
    WITH TABLE public.events_pgaa_1_prt_p202501_write;
```

ALTER TABLE events_pgaa ATTACH PARTITION public.events_pgaa_1_prt_p202501_read
  FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

-- ③ 로컬 파티션 디스크 회수 (검증 후 드롭)
DROP TABLE public.events_pgaa_1_prt_p202501;
```

## 3. S3에 있는 데이터를 다시 부모 테이블에 파티션으로 연결하기

핵심은 **DETACH → ATTACH 교체(exchange) 패턴**입니다. WarehousePG 공식 문서의 "Exchanging a Partition"과 동일한 방식으로, 이번엔 로컬 heap 테이블 대신 PGAA(S3) 테이블을 붙입니다.

```sql
-- ① S3에 있는 기존 데이터를 가리키는 PGAA 테이블 생성 (읽기 전용)
CREATE TABLE sales_2024_01_cold ()
USING PGAA
WITH (
    pgaa.storage_location = 'sales_cold_tier',
    pgaa.path = 'sales_2024_01'
);

-- ② 부모 파티션 테이블에 다시 붙이기
ALTER TABLE sales ATTACH PARTITION sales_2024_01_cold
  FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');
```

이제 `sales`를 조회하면 로컬 heap 파티션(핫 데이터)과 S3의 PGAA 파티션(콜드 데이터)이 하나의 쿼리로 함께 조회됩니다. PGAA가 지원하는 연산은 Seafowl(DirectScan)로 오프로딩되고, 로컬 heap과의 조인이 섞이면 CompatScan으로 자동 전환됩니다.

## 주의할 제약사항

- **PGAA로 매핑된 파티션은 읽기 전용**입니다. 다시 쓰기가 필요하면 해당 파티션만 `ALTER TABLE ... SET ACCESS METHOD heap`으로 되돌리거나, 새 heap 테이블로 교체(exchange)해야 합니다.
- 외부/이종 스토리지 파티션(leaf partition)에는 `INSERT/UPDATE/DELETE/TRUNCATE`, 일부 `ALTER TABLE ... ALTER PARTITION`(서브파티션 템플릿, 분산 정책 변경 등)이 제한됩니다.
- 파티션 교체(exchange)는 계층 **최하위 leaf 파티션**에서만 가능하며, 파티션된 테이블끼리는 교체할 수 없습니다.
- PGAA on WarehousePG는 GPORCA 옵티마이저를 쓰지 않고 표준 Postgres 플래너만 사용합니다.
- `gpbackup`은 leaf 파티션이 external/foreign(PGAA 포함) 테이블이면 그 파티션의 데이터를 백업하지 않습니다 — 별도 백업 전략이 필요합니다.

## 참고: 완전 자동화가 필요하다면

수동 DETACH/CTAS/ATTACH 대신 나이 기준으로 자동으로 파티션을 만들고 콜드 티어로 넘기는 걸 원하면, 이는 **PGD의 AutoPartition + PGAA** 조합(`pgaa.convert_to_tiered_table()`)에서 지원됩니다. 다만 이는 WarehousePG(Greenplum 계열 MPP 엔진) 단독이 아니라 PGD 클러스터 기능이라, 순수 WarehousePG 환경이라면 위의 수동 패턴이 맞는 접근입니다.
