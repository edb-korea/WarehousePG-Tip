-- 0. 스키마
CREATE SCHEMA IF NOT EXISTS redhat;
COMMENT ON SCHEMA redhat IS 'OS(Red Hat) 레벨 정보를 External Web Table로 조회';

-- 1. 재실행을 위한 정리 (의존성 순서: 뷰 -> 테이블)
DROP VIEW IF EXISTS redhat.v_svc_status;
DROP EXTERNAL TABLE IF EXISTS redhat.ext_svc_status_segments;
DROP EXTERNAL TABLE IF EXISTS redhat.ext_svc_status_coordinator;

-- 2. 세그먼트 호스트용 (호스트당 1회 실행)
CREATE EXTERNAL WEB TABLE redhat.ext_svc_status_segments (
    hostname      text,
    unit_name     text,
    load_state    text,
    active_state  text,
    sub_state     text,
    enabled_state text,
    description   text
)
EXECUTE $$
UF=$(/usr/bin/systemctl list-unit-files --type=service --no-pager --no-legend --plain 2>/dev/null)
UF="$UF" /usr/bin/systemctl list-units --type=service --all --no-pager --no-legend --plain 2>/dev/null | awk -v h="$(hostname)" '
BEGIN {
  n = split(ENVIRON["UF"], L, "\n")
  for (i = 1; i <= n; i++) { split(L[i], a, " "); en[a[1]] = a[2] }
}
{
  d = ""
  for (i = 5; i <= NF; i++) d = d (i > 5 ? " " : "") $i
  gsub(/\|/, "/", d)
  e = ($1 in en) ? en[$1] : "n/a"
  printf "%s|%s|%s|%s|%s|%s|%s\n", h, $1, $2, $3, $4, e, d
}'
exit 0
$$
ON HOST
FORMAT 'TEXT' (DELIMITER '|' ESCAPE 'OFF');

-- 3. 코디네이터용
CREATE EXTERNAL WEB TABLE redhat.ext_svc_status_coordinator (
    hostname      text,
    unit_name     text,
    load_state    text,
    active_state  text,
    sub_state     text,
    enabled_state text,
    description   text
)
EXECUTE $$
UF=$(/usr/bin/systemctl list-unit-files --type=service --no-pager --no-legend --plain 2>/dev/null)
UF="$UF" /usr/bin/systemctl list-units --type=service --all --no-pager --no-legend --plain 2>/dev/null | awk -v h="$(hostname)" '
BEGIN {
  n = split(ENVIRON["UF"], L, "\n")
  for (i = 1; i <= n; i++) { split(L[i], a, " "); en[a[1]] = a[2] }
}
{
  d = ""
  for (i = 5; i <= NF; i++) d = d (i > 5 ? " " : "") $i
  gsub(/\|/, "/", d)
  e = ($1 in en) ? en[$1] : "n/a"
  printf "%s|%s|%s|%s|%s|%s|%s\n", h, $1, $2, $3, $4, e, d
}'
exit 0
$$
ON COORDINATOR          -- v6는 ON MASTER
FORMAT 'TEXT' (DELIMITER '|' ESCAPE 'OFF');

-- 4. 통합 뷰
CREATE VIEW redhat.v_svc_status AS
SELECT 'coordinator'::text AS role, * FROM redhat.ext_svc_status_coordinator
UNION ALL
SELECT 'segment'::text     AS role, * FROM redhat.ext_svc_status_segments;
